declare(strict_types=1);

namespace BaksDev\Services\Entity\Schedule;

use BaksDev\Core\Entity\EntityState;
use BaksDev\Services\Entity\Event\ServiceEvent;
use BaksDev\Services\Type\Schedule\ServiceScheduleUid;
use DateTimeImmutable;
use Doctrine\DBAL\Types\Types;
use Doctrine\ORM\Mapping as ORM;
use InvalidArgumentException;
use Symfony\Component\Validator\Constraints as Assert;

#[ORM\Entity]
#[ORM\Table(name: 'service_schedule')]
class ServiceSchedule extends EntityState
{
    /**
     * Идентификатор сущности
     */
    #[Assert\NotBlank]
    #[Assert\Uuid]
    #[ORM\Id]
    #[ORM\Column(type: ServiceScheduleUid::TYPE)]
    private ServiceScheduleUid $id;

    #[ORM\Id]
    #[ORM\ManyToOne(targetEntity: ServiceEvent::class, inversedBy: 'schedule')]
    #[ORM\JoinColumn(name: 'event', referencedColumnName: 'id')]
    private ServiceEvent $event;

    /** Дата услуги */
    #[Assert\NotBlank]
    #[ORM\Column(type: Types::DATETIME_IMMUTABLE, nullable: false)]
    private DateTimeImmutable $date;

    /** Слот */
    #[Assert\NotBlank]
    #[ORM\Column(type: Types::INTEGER, nullable: false, options: ['default' => 0])]
    private int $slot = 0;

    public function __construct(ServiceEvent $event)
    {
        $this->id = new ServiceScheduleUid();
        $this->event = $event;
    }

    public function getId(): ServiceScheduleUid
    {
        return $this->id;
    }

    public function getDto($dto): mixed
    {
        $dto = is_string($dto) && class_exists($dto) ? new $dto() : $dto;

        if($dto instanceof ServiceScheduleInterface)
        {
            return parent::getDto($dto);
        }

        throw new InvalidArgumentException(sprintf('Class %s interface error', $dto::class));
    }

    public function setEntity($dto): mixed
    {
        if($dto instanceof ServiceScheduleInterface || $dto instanceof self)
        {
            return parent::setEntity($dto);
        }

        throw new InvalidArgumentException(sprintf('Class %s interface error', $dto::class));
    }
}
